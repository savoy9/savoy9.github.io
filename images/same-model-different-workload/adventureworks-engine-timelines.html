14-query plan: 12-month trend

Returns the native 12-month trend title and its data rows in two separate result tables.

Connector output: 1 measured row × 1 column in result 1 of 2; result 2 (chart body) is not exposed by the connector.

Returned columns: [AW Trend Title]

Executed DAX (formatted by SQLBI)

DEFINE
    MEASURE 'FactInternetSales'[AW Trend Title] =
        (
            /* USER DAX BEGIN */
            VAR hi = [Source Last Date]
            VAR lo =
                EDATE ( DATE ( YEAR ( hi ), MONTH ( hi ), 1 ), -11 )
            VAR n =
                CALCULATE (
                    [Order Lines],
                    FILTER (
                        ALL ( DimDate ),
                        DimDate[FullDateAlternateKey] >= lo
                            && DimDate[FullDateAlternateKey] <= EOMONTH ( hi, 0 )
                    )
                )
            VAR coverage =
                CALCULATE (
                    [Partial Month Text],
                    REMOVEFILTERS ( DimDate ),
                    TREATAS ( { FORMAT ( hi, "yyyy-MM" ) }, DimDate[YearMonth] )
                )
            RETURN
                "12-month trend · " & FORMAT ( lo, "MMM yyyy" ) & "–"
                    & FORMAT ( hi, "MMM yyyy" )
                    & IF (
                        COALESCE ( n, 0 ) = 0,
                        " · No data for the current filters — clear Where/What or Reset",
                        ""
                    )
                    & IF ( NOT ISBLANK ( coverage ), " · " & coverage, "" ) /* USER DAX END */
        )
    VAR __ValueFilterDM0 =
        FILTER (
            KEEPFILTERS (
                SUMMARIZECOLUMNS (
                    'DimDate'[YearMonth],
                    'DimDate'[YearMonthKey],
                    "Trend_Window", IGNORE ( 'FactInternetSales'[Trend Window] )
                )
            ),
            [Trend_Window] = 1
        )
    VAR __DS0Core =
        SUMMARIZECOLUMNS (
            'DimDate'[YearMonth],
            'DimDate'[YearMonthKey],
            __ValueFilterDM0,
            "Total_Sales", 'FactInternetSales'[Total Sales],
            "Order_Count", IGNORE ( 'FactInternetSales'[Order Count] ),
            "Gross_Margin", IGNORE ( 'FactInternetSales'[Gross Margin] ),
            "Partial_Month_Text", IGNORE ( 'FactInternetSales'[Partial Month Text] )
        )
    VAR __DS0PrimaryShowAll =
        ADDMISSINGITEMS ( 'DimDate'[YearMonth],
        'DimDate'[YearMonthKey],
        __DS0Core,
        'DimDate'[YearMonth],
        'DimDate'[YearMonthKey],
        __ValueFilterDM0 )
    VAR __DS0PrimaryWindowed =
        TOPN (
            1001,
            __DS0PrimaryShowAll,
            'DimDate'[YearMonthKey], 1,
            'DimDate'[YearMonth], 1
        )

EVALUATE
SUMMARIZECOLUMNS (
    __ValueFilterDM0,
    "AW Trend Title", IGNORE ( 'FactInternetSales'[AW Trend Title] )
)

EVALUATE
__DS0PrimaryWindowed
ORDER BY
    'DimDate'[YearMonthKey],
    'DimDate'[YearMonth]
